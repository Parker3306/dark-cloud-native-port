cmake_minimum_required(VERSION 3.24)
project(dccore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

# dccore: the engine-independent part of the port (file formats, game rules, simulation). No Godot, no graphics, no audio.
add_library(dccore STATIC)
target_include_directories(dccore PUBLIC include)
file(GLOB_RECURSE DCCORE_SRC CONFIGURE_DEPENDS src/*.cpp)
target_sources(dccore PRIVATE ${DCCORE_SRC})

add_executable(dcgolden tools/dcgolden.cpp)
target_link_libraries(dcgolden PRIVATE dccore)
if(WIN32)
  target_link_options(dcgolden PRIVATE -static)  # standalone exe: no llvm-mingw runtime DLLs needed on PATH
endif()

enable_testing()
file(GLOB DCCORE_TESTS CONFIGURE_DEPENDS tests/*.cpp)
foreach(t ${DCCORE_TESTS})
  get_filename_component(n ${t} NAME_WE)
  add_executable(${n} ${t})
  target_link_libraries(${n} PRIVATE dccore)
  add_test(NAME ${n} COMMAND ${n})
endforeach()
